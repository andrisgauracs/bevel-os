# Bevel OS desktop simulator (Linux): the OS sources of main/ built for the host, with the
# drivers of host/. See host/README.md.
#
#   cmake -S host -B build-host && cmake --build build-host -j && build-host/bevel-sim
cmake_minimum_required(VERSION 3.16)
project(bevel_sim C ASM)

set(BEVEL_VNC ON CACHE BOOL "Build the VNC server in")
set(BEVEL_VNC_PORT 5900 CACHE STRING "VNC TCP port")
set(BEVEL_VNC_PASSWORD "" CACHE STRING "VNC password (empty: no authentication)")
set(BEVEL_DOOM ON CACHE BOOL "Build Doom for the simulator's drive")
set(BEVEL_VNC_PUBLIC OFF CACHE BOOL "Accept VNC viewers from other machines (default: 127.0.0.1 only)")

set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(MAIN ${ROOT}/main)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# ---------------------------------------------------------------- SDL2
# The system's development package if there is one, else headers unpacked by host/get-sdl2.sh
# (no root needed) next to the runtime library that desktop systems already have.
find_package(PkgConfig QUIET)
if(PKG_CONFIG_FOUND)
    pkg_check_modules(SDL2 QUIET sdl2)
endif()
if(NOT SDL2_FOUND)
    set(SDL2_LOCAL ${ROOT}/build-host/deps/root/usr/include/SDL2)
    find_path(SDL2_INCLUDE_DIRS SDL.h PATHS ${SDL2_LOCAL} NO_DEFAULT_PATH)
    find_library(SDL2_LIBRARIES NAMES SDL2-2.0 SDL2 libSDL2-2.0.so.0)
    if(NOT SDL2_INCLUDE_DIRS OR NOT SDL2_LIBRARIES)
        message(FATAL_ERROR "SDL2 not found: install libsdl2-dev, or run host/get-sdl2.sh")
    endif()
    # SDL_config.h of the Debian package lives in a multiarch folder
    list(APPEND SDL2_INCLUDE_DIRS ${ROOT}/build-host/deps/root/usr/include/x86_64-linux-gnu)
endif()

# ---------------------------------------------------------------- configuration
set(CONFIG_H ${CMAKE_BINARY_DIR}/config/sdkconfig.h)
file(WRITE ${CONFIG_H}.in "// Generated by host/CMakeLists.txt: the Kconfig options of the simulator build\n#pragma once\n#define CONFIG_BEVEL_DISPLAY_NONE 1\n")
if(BEVEL_VNC)
    file(APPEND ${CONFIG_H}.in "#define CONFIG_BEVEL_VNC 1\n#define CONFIG_BEVEL_VNC_PORT ${BEVEL_VNC_PORT}\n#define CONFIG_BEVEL_VNC_PASSWORD \"${BEVEL_VNC_PASSWORD}\"\n")
    if(NOT BEVEL_VNC_PUBLIC)
        file(APPEND ${CONFIG_H}.in "#define CONFIG_BEVEL_VNC_LOCALHOST_ONLY 1\n")
    endif()
endif()
file(APPEND ${CONFIG_H}.in "#define CONFIG_BEVEL_WIFI_SSID \"\"\n#define CONFIG_BEVEL_WIFI_PASSWORD \"\"\n")
configure_file(${CONFIG_H}.in ${CONFIG_H} COPYONLY)

# ---------------------------------------------------------------- apps for the drive
# The same apps/*.c, as x86-64 shared objects (named .elf like on the board).
set(DISK_APPS ${CMAKE_BINARY_DIR}/apps)
file(MAKE_DIRECTORY ${DISK_APPS})
set(APP_FLAGS -O2 -g -fPIC -shared -Wl,-Bsymbolic -Wall -I${ROOT}/apps)
set(APP_ELFS)
foreach(app Snake Hello CrashTest)
    add_custom_command(OUTPUT ${DISK_APPS}/${app}.elf
        COMMAND ${CMAKE_C_COMPILER} ${APP_FLAGS} ${ROOT}/apps/crt.c ${ROOT}/apps/${app}.c -o ${DISK_APPS}/${app}.elf
        DEPENDS ${ROOT}/apps/crt.c ${ROOT}/apps/${app}.c ${ROOT}/apps/bevel.h
        COMMENT "App ${app}.elf (host)")
    list(APPEND APP_ELFS ${DISK_APPS}/${app}.elf)
endforeach()

set(DOOM_ELF ${DISK_APPS}/Doom.elf)
if(BEVEL_DOOM)
    set(DG ${ROOT}/third_party/doomgeneric/doomgeneric)
    set(DOOM_FILES dummy am_map doomdef doomstat dstrings d_event d_items d_iwad d_loop d_main d_mode d_net f_finale f_wipe g_game
        hu_lib hu_stuff info i_cdmus i_endoom i_joystick i_scale i_sound i_system i_timer memio m_argv m_bbox m_cheat m_config
        m_controls m_fixed m_menu m_misc m_random p_ceilng p_doors p_enemy p_floor p_inter p_lights p_map p_maputl p_mobj
        p_plats p_pspr p_saveg p_setup p_sight p_spec p_switch p_telept p_tick p_user r_bsp r_data r_draw r_main r_plane
        r_segs r_sky r_things sha1 sounds statdump st_lib st_stuff s_sound tables v_video wi_stuff w_checksum w_file w_main
        w_wad z_zone w_file_stdc i_input i_video doomgeneric)
    set(DOOM_SRCS ${ROOT}/apps/crt.c ${ROOT}/apps/doom/doom_bevel.c)
    foreach(f ${DOOM_FILES})
        list(APPEND DOOM_SRCS ${DG}/${f}.c)
    endforeach()
    add_library(doom_app MODULE ${DOOM_SRCS})
    target_compile_options(doom_app PRIVATE -O2 -w -fno-strict-aliasing)
    target_compile_definitions(doom_app PRIVATE CMAP256 DOOMGENERIC_RESX=320 DOOMGENERIC_RESY=200 NORMALUNIX _DEFAULT_SOURCE)
    target_include_directories(doom_app PRIVATE ${DG} ${ROOT}/apps)
    target_link_options(doom_app PRIVATE -Wl,-Bsymbolic)
    set_target_properties(doom_app PROPERTIES PREFIX "" OUTPUT_NAME Doom SUFFIX ".elf" LIBRARY_OUTPUT_DIRECTORY ${DISK_APPS})
else()
    file(WRITE ${DOOM_ELF} "")
endif()

# The OS keeps its bundled files in the firmware image and reinstalls them on the drive
# (storage.c); here they are linked into the executable the same way.
set(EMBED_S ${CMAKE_BINARY_DIR}/embed.S)
set(EMBED "")
foreach(pair "Welcome_txt:${ROOT}/disk/Documents/Welcome.txt" "Bevel_bmp:${ROOT}/disk/Pictures/Bevel.bmp"
             "Snake_elf:${DISK_APPS}/Snake.elf" "Hello_elf:${DISK_APPS}/Hello.elf"
             "CrashTest_elf:${DISK_APPS}/CrashTest.elf" "Doom_elf:${DOOM_ELF}")
    string(REPLACE ":" ";" kv ${pair})
    list(GET kv 0 sym)
    list(GET kv 1 file)
    string(APPEND EMBED "    .global _binary_${sym}_start\n    .global _binary_${sym}_end\n    .balign 16\n_binary_${sym}_start:\n    .incbin \"${file}\"\n_binary_${sym}_end:\n    .byte 0\n")
endforeach()
file(WRITE ${EMBED_S} "    .section .rodata\n${EMBED}    .section .note.GNU-stack,\"\",@progbits\n")
set_source_files_properties(${EMBED_S} PROPERTIES OBJECT_DEPENDS "${APP_ELFS};${DOOM_ELF}")

# ---------------------------------------------------------------- the OS
set(OS_SRCS main.c cursor.c headless.c gfx.c comp.c input.c serial.c perf.c wm.c ui.c menu.c shell.c dlg.c apps.c
    apps_basic.c storage.c app_loader.c app_files.c app_editor.c app_misc.c app_net.c app_browser.c vnc.c vnc_des.c
    assets/assets_gen.c)
list(TRANSFORM OS_SRCS PREPEND ${MAIN}/)
add_executable(bevel-sim ${OS_SRCS} sim.c rtos.c hal.c fswrap.c app_dl.c net_host.c ${EMBED_S})
add_custom_target(sim_apps DEPENDS ${APP_ELFS})
add_dependencies(bevel-sim sim_apps)
if(BEVEL_DOOM)
    add_dependencies(bevel-sim doom_app)
endif()
target_include_directories(bevel-sim BEFORE PRIVATE ${CMAKE_BINARY_DIR}/config ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(bevel-sim PRIVATE ${MAIN} ${MAIN}/assets ${CMAKE_CURRENT_SOURCE_DIR} ${SDL2_INCLUDE_DIRS})
target_compile_definitions(bevel-sim PRIVATE BEVEL_HOST=1 _GNU_SOURCE BEVEL_DEFAULT_DISK="${CMAKE_BINARY_DIR}/disk")
target_compile_options(bevel-sim PRIVATE -Wall -Wno-format-truncation -Wno-unused-function -fno-strict-aliasing)
target_link_libraries(bevel-sim PRIVATE ${SDL2_LIBRARIES} pthread dl
    -Wl,--wrap=fopen,--wrap=opendir,--wrap=stat,--wrap=mkdir,--wrap=rmdir,--wrap=unlink,--wrap=remove,--wrap=rename)
